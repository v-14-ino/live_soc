    52→function formatTime(ts: string): string {
    53→  try {
    54→    const d = new Date(ts);
    55→    if (isNaN(d.getTime())) return "--:--:--";
    56→    return d.toLocaleTimeString("en-GB", { hour12: false });
    57→  } catch {
    58→    return "--:--:--";
    59→  }
    60→}
    61→
    62→function formatDateTime(ts: string): string {
    63→  try {
    64→    const d = new Date(ts);
    65→    if (isNaN(d.getTime())) return "----/--/-- --:--:--";
    66→    const pad = (n: number) => String(n).padStart(2, "0");
    67→    return `${d.getFullYear()}/${pad(d.getMonth() + 1)}/${pad(d.getDate())} ${pad(d.getHours())}:${pad(d.getMinutes())}:${pad(d.getSeconds())}`;
    68→  } catch {
    69→    return "----/--/-- --:--:--";
    70→  }
    71→}
    72→
    73→function durationLabel(first: string, last: string): string {
    74→  try {
    75→    const a = new Date(first).getTime();
    76→    const b = new Date(last).getTime();
    77→    if (!isFinite(a) || !isFinite(b)) return "—";
    78→    const sec = Math.max(0, Math.round((b - a) / 1000));
    79→    if (sec < 60) return `${sec}s`;
    80→    const m = Math.floor(sec / 60);
    81→    const s = sec % 60;
    82→    if (m < 60) return `${m}m ${s}s`;
    83→    const h = Math.floor(m / 60);
    84→    return `${h}h ${m % 60}m`;
    85→  } catch {
    86→    return "—";
    87→  }
    88→}

    90→// attackPath: live/historical both store as "A -> B -> C" string; tolerate arrays.
    91→function parseAttackPath(ap: string | string[] | null | undefined): string[] {
    92→  if (ap == null) return [];
    93→  if (Array.isArray(ap)) return ap.map(String).filter(Boolean);
    94→  const s = String(ap).trim();
    95→  if (!s) return [];
    96→  if (s.includes("->")) return s.split("->").map((x) => x.trim()).filter(Boolean);
    97→  if (s.includes("\n")) return s.split("\n").map((x) => x.trim()).filter(Boolean);
    98→  if (s.includes(",")) return s.split(",").map((x) => x.trim()).filter(Boolean);
    99→  return [s];
   100→}
   101→
   102→// relatedEventIds: live = real array; historical (raw Prisma row) = comma string.
   103→function parseRelatedEventIds(
   104→  ids: string | string[] | null | undefined,
   105→): string[] {
   106→  if (ids == null) return [];
   107→  if (Array.isArray(ids)) return ids.map(String).filter(Boolean);
   108→  const s = String(ids).trim();
   109→  if (!s) return [];
   110→  return s.split(",").map((x) => x.trim()).filter(Boolean);
   111→}

   157→function severityRank(s: Severity): number {
   158→  return SEVERITY_ORDER[s] ?? 0;

   161→function sortScenarios(list: OffenseScenario[]): OffenseScenario[] {
   162→  return list.slice().sort((a, b) => {
   163→    const sev = severityRank(b.severity) - severityRank(a.severity);
   164→    if (sev !== 0) return sev;
   165→    const ta = new Date(a.lastObserved).getTime() || 0;
   166→    const tb = new Date(b.lastObserved).getTime() || 0;
   167→    return tb - ta;
   168→  });
   169→}

   903→export function OffenseView() {
   904→  const sessionId = useAppStore((s) => s.sessionId);
   905→  const historicalSession = useAppStore((s) => s.historicalSession);
   906→  const liveScenarios = useAppStore((s) => s.offenseScenarios);
   907→  const setOffenseScenarios = useAppStore((s) => s.setOffenseScenarios);
   908→
   909→  // Determine mode
   910→  const mode: "live" | "historical" | "empty" =
   911→    historicalSession != null
   912→      ? "historical"
   913→      : sessionId != null
   914→        ? "live"
   915→        : "empty";
   916→
   917→  // Historical state
   918→  const [histScenarios, setHistScenarios] = useState<OffenseScenario[]>([]);
   919→  const [histEvents, setHistEvents] = useState<SecurityEvent[]>([]);
   920→  const [histLoading, setHistLoading] = useState(false);
   921→  const [histError, setHistError] = useState<string | null>(null);
   922→
   923→  // Live seed state
   924→  const [liveSeeding, setLiveSeeding] = useState(false);
   925→
   926→  // Dialog state
   927→  const [dialogScenarioId, setDialogScenarioId] = useState<string | null>(null);
   928→  const [dialogEvents, setDialogEvents] = useState<SecurityEvent[]>([]);
   929→  const [dialogEventsLoading, setDialogEventsLoading] = useState(false);
   930→
   931→  // ---------------- Historical fetch ----------------
   932→  useEffect(() => {
   933→    if (mode !== "historical" || !historicalSession) return;
   934→    let cancelled = false;
   935→    const run = async () => {
   936→      setHistLoading(true);
   937→      setHistError(null);
   938→      try {
   939→        const res = await api.getHistoryDetail(historicalSession.id);
   940→        if (cancelled) return;
   941→        setHistScenarios((res.offenseScenarios ?? []).map(normalizeScenario));
   942→        setHistEvents(res.events ?? []);
   943→      } catch (e: unknown) {
   944→        if (cancelled) return;
   945→        const msg =
   946→          e instanceof Error ? e.message : "Failed to load historical session.";
   947→        setHistError(msg);
   948→        toast.error(msg);
   949→      } finally {
   950→        if (!cancelled) setHistLoading(false);
   951→      }
   952→    };
   953→    void run();
   954→    return () => {
   955→      cancelled = true;
   956→    };
   957→  }, [mode, historicalSession]);
   958→
   959→  // ---------------- Live seed fetch ----------------
   960→  useEffect(() => {
   961→    if (mode !== "live" || !sessionId) return;
   962→    let cancelled = false;
   963→    const run = async () => {
   964→      setLiveSeeding(true);
   965→      try {
   966→        const res = await api.getScenarios(sessionId);
   967→        if (cancelled) return;
   968→        const offense = (res.offense ?? []).map(normalizeScenario);
   969→        // Replace the live scenario list with the authoritative snapshot.
   970→        // Any WS updates that arrive afterward will upsert on top.
   971→        setOffenseScenarios(offense);
   972→      } catch (e: unknown) {
   973→        if (cancelled) return;
   974→        // Silent — store may already have streamed scenarios via WS.
   975→        console.warn("[OffenseView] failed to seed scenarios:", e);
   976→      } finally {
   977→        if (!cancelled) setLiveSeeding(false);
   978→      }
   979→    };
   980→    void run();
   981→    return () => {
   982→      cancelled = true;
   983→    };
   984→  }, [mode, sessionId, setOffenseScenarios]);
   985→
   986→  // ---------------- Dialog events fetch (live mode only) ----------------
   987→  const dialogScenario = useMemo<OffenseScenario | null>(() => {
   988→    if (!dialogScenarioId) return null;
   989→    const pool = mode === "historical" ? histScenarios : liveScenarios;
   990→    return pool.find((s) => s.scenarioId === dialogScenarioId) ?? null;
   991→  }, [dialogScenarioId, mode, histScenarios, liveScenarios]);
   992→
   993→  useEffect(() => {
   994→    if (!dialogScenarioId || mode !== "live" || !sessionId) return;
   995→    let cancelled = false;
   996→    const run = async () => {
   997→      setDialogEventsLoading(true);
   998→      setDialogEvents([]);
   999→      try {
  1000→        const res = await api.getEvents(sessionId, 500);
  1001→        if (cancelled) return;
  1002→        setDialogEvents(res.events ?? []);
  1003→      } catch (e: unknown) {
  1004→        if (cancelled) return;
  1005→        console.warn("[OffenseView] failed to load events for dialog:", e);
  1006→      } finally {
  1007→        if (!cancelled) setDialogEventsLoading(false);
  1008→      }
  1009→    };
  1010→    void run();
  1011→    return () => {
  1012→      cancelled = true;
  1013→    };
  1014→  }, [dialogScenarioId, mode, sessionId]);
  1015→
  1016→  // ---------------- Active scenarios (sorted) ----------------
  1017→  const scenarios = useMemo<OffenseScenario[]>(() => {
  1018→    const raw = mode === "historical" ? histScenarios : liveScenarios;
  1019→    return sortScenarios(raw.map(normalizeScenario));
  1020→  }, [mode, histScenarios, liveScenarios]);
  1021→
  1022→  const dialogOpen = !!dialogScenarioId;
  1023→  const dialogEventsForMode: SecurityEvent[] =
  1024→    mode === "historical" ? histEvents : dialogEvents;
  1025→
  1026→  // ---------------- Empty mode ----------------
  1027→  if (mode === "empty") {
  1028→    return (
  1029→      <div className="flex h-full flex-col overflow-hidden">
  1030→        <OffenseHeader mode="empty" subtitle="" />
  1031→        <div className="flex-1 overflow-y-auto">
  1032→          <NoSessionEmptyState />
  1033→        </div>
  1034→      </div>
  1035→    );
  1036→  }
  1037→
  1038→  const sessionMode: "live" | "historical" = mode;
  1039→  const subtitle =
  1040→    mode === "live"
  1041→      ? `Live session · ${scenarios.length} potential scenario${scenarios.length === 1 ? "" : "s"} under analysis`
  1042→      : historicalSession
  1043→        ? `Historical session · ${historicalSession.targetAddress} · ${historicalSession.startedAt ? new Date(historicalSession.startedAt).toLocaleString() : ""}`
  1044→        : "";
  1045→
  1046→  const isLoading =
  1047→    (mode === "historical" && histLoading) ||
  1048→    (mode === "live" && liveSeeding && scenarios.length === 0);
  1049→
  1050→  return (
  1051→    <div className="flex h-full flex-col overflow-hidden">
  1052→      <OffenseHeader mode={sessionMode} subtitle={subtitle} />
  1053→
  1054→      {/* Summary row */}
  1055→      <div className="shrink-0 border-b border-border/40 px-3 py-3">
  1056→        <SummaryRow scenarios={scenarios} />
  1057→      </div>
  1058→
  1059→      {/* Scrollable body */}
  1060→      <div className="soc-scrollbar min-h-0 flex-1 overflow-y-auto overflow-x-hidden">
  1061→        <div className="flex flex-col gap-3 p-3">
  1062→          {/* Section header */}
  1063→          <div className="flex items-center justify-between gap-2">
  1064→            <div className="flex items-center gap-2">
  1065→              <h3 className="font-mono-data text-xs font-semibold uppercase tracking-wider text-foreground">
  1066→                Potential Attack Scenarios
  1067→              </h3>
  1068→              <span className="rounded-sm border border-[color:var(--soc-critical)]/40 bg-[color:var(--soc-critical)]/10 px-1.5 py-0.5 font-mono-data text-[10px] font-bold text-[color:var(--soc-critical)]">
  1069→                {scenarios.length}
  1070→              </span>
  1071→            </div>
  1072→            {mode === "historical" && histError && (
  1073→              <span className="font-mono-data text-[10px] text-[color:var(--soc-critical)]">
  1074→                ERROR: {histError}
  1075→              </span>
  1076→            )}
  1077→          </div>
  1078→
  1079→          {/* Cards grid */}
  1080→          {isLoading ? (
  1081→            <div className="grid grid-cols-1 gap-3 xl:grid-cols-2">
  1082→              {[0, 1, 2, 3].map((i) => (
  1083→                <ScenarioCardSkeleton key={i} />
  1084→              ))}
  1085→            </div>
  1086→          ) : scenarios.length === 0 ? (
  1087→            <NoScenariosEmptyState mode={sessionMode} />
  1088→          ) : (
  1089→            <div className="grid grid-cols-1 gap-3 xl:grid-cols-2">
