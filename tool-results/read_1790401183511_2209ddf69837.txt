   903→   903→export function OffenseView() {
   904→   904→  const sessionId = useAppStore((s) => s.sessionId);
   905→   905→  const historicalSession = useAppStore((s) => s.historicalSession);
   906→   906→  const liveScenarios = useAppStore((s) => s.offenseScenarios);
   907→   907→  const setOffenseScenarios = useAppStore((s) => s.setOffenseScenarios);
   908→   908→  // Cross-view link target: another view (e.g. Defense) may set this to a
   909→   909→  // scenarioId to request that the Offense view auto-open that scenario's
   910→   910→  // detail dialog. We sync it into our local dialogScenarioId state below.
   911→   911→  const selectedOffenseId = useAppStore((s) => s.selectedOffenseId);
   912→   912→  const setSelectedOffense = useAppStore((s) => s.setSelectedOffense);
   913→   913→
   914→   914→  // Determine mode
   915→   915→  const mode: "live" | "historical" | "empty" =
   916→   916→    historicalSession != null
   917→   917→      ? "historical"
   918→   918→      : sessionId != null
   919→   919→        ? "live"
   920→   920→        : "empty";
   921→   921→
   922→   922→  // Historical state
   923→   923→  const [histScenarios, setHistScenarios] = useState<OffenseScenario[]>([]);
   924→   924→  const [histEvents, setHistEvents] = useState<SecurityEvent[]>([]);
   925→   925→  const [histLoading, setHistLoading] = useState(false);
   926→   926→  const [histError, setHistError] = useState<string | null>(null);
   927→   927→
   928→   928→  // Live seed state
   929→   929→  const [liveSeeding, setLiveSeeding] = useState(false);
   930→   930→
   931→   931→  // Dialog state
   932→   932→  const [dialogScenarioId, setDialogScenarioId] = useState<string | null>(null);
   933→   933→  const [dialogEvents, setDialogEvents] = useState<SecurityEvent[]>([]);
   934→   934→  const [dialogEventsLoading, setDialogEventsLoading] = useState(false);
   935→   935→
   936→   936→  // ---------------- Historical fetch ----------------
   937→   937→  useEffect(() => {
   938→   938→    if (mode !== "historical" || !historicalSession) return;
   939→   939→    let cancelled = false;
   940→   940→    const run = async () => {
   941→   941→      setHistLoading(true);
   942→   942→      setHistError(null);
   943→   943→      try {
   944→   944→        const res = await api.getHistoryDetail(historicalSession.id);
   945→   945→        if (cancelled) return;
   946→   946→        setHistScenarios((res.offenseScenarios ?? []).map(normalizeScenario));
   947→   947→        setHistEvents(res.events ?? []);
   948→   948→      } catch (e: unknown) {
   949→   949→        if (cancelled) return;
   950→   950→        const msg =
   951→   951→          e instanceof Error ? e.message : "Failed to load historical session.";
   952→   952→        setHistError(msg);
   953→   953→        toast.error(msg);
   954→   954→      } finally {
   955→   955→        if (!cancelled) setHistLoading(false);
   956→   956→      }
   957→   957→    };
   958→   958→    void run();
   959→   959→    return () => {
   960→   960→      cancelled = true;
   961→   961→    };
   962→   962→  }, [mode, historicalSession]);
   963→   963→
   964→   964→  // ---------------- Live seed fetch ----------------
   965→   965→  useEffect(() => {
   966→   966→    if (mode !== "live" || !sessionId) return;
   967→   967→    let cancelled = false;
   968→   968→    const run = async () => {
   969→   969→      setLiveSeeding(true);
   970→   970→      try {
   971→   971→        const res = await api.getScenarios(sessionId);
   972→   972→        if (cancelled) return;
   973→   973→        const offense = (res.offense ?? []).map(normalizeScenario);
   974→   974→        // Replace the live scenario list with the authoritative snapshot.
   975→   975→        // Any WS updates that arrive afterward will upsert on top.
   976→   976→        setOffenseScenarios(offense);
   977→   977→      } catch (e: unknown) {
   978→   978→        if (cancelled) return;
   979→   979→        // Silent — store may already have streamed scenarios via WS.
   980→   980→        console.warn("[OffenseView] failed to seed scenarios:", e);
   981→   981→      } finally {
   982→   982→        if (!cancelled) setLiveSeeding(false);
   983→   983→      }
   984→   984→    };
   985→   985→    void run();
   986→   986→    return () => {
   987→   987→      cancelled = true;
   988→   988→    };
   989→   989→  }, [mode, sessionId, setOffenseScenarios]);
   990→   990→
   991→   991→  // ---------------- Cross-view link: Defense → Offense ----------------
   992→   992→  // When another view sets `selectedOffenseId` to a scenarioId, mirror it
   993→   993→  // into our local dialogScenarioId so the detail dialog opens for that
   994→   994→  // scenario. Cleared when the dialog closes (see onOpenChange below).
   995→   995→  useEffect(() => {
   996→   996→    if (selectedOffenseId) {
   997→   997→      setDialogScenarioId(selectedOffenseId);
   998→   998→    }
   999→   999→  }, [selectedOffenseId]);
  1000→  1000→
  1001→  1001→  // ---------------- Dialog events fetch (live mode only) ----------------
  1002→  1002→  const dialogScenario = useMemo<OffenseScenario | null>(() => {
  1003→  1003→    if (!dialogScenarioId) return null;
  1004→  1004→    const pool = mode === "historical" ? histScenarios : liveScenarios;
  1005→  1005→    return pool.find((s) => s.scenarioId === dialogScenarioId) ?? null;
  1006→  1006→  }, [dialogScenarioId, mode, histScenarios, liveScenarios]);
  1007→  1007→
  1008→  1008→  useEffect(() => {
  1009→  1009→    if (!dialogScenarioId || mode !== "live" || !sessionId) return;
  1010→  1010→    let cancelled = false;
  1011→  1011→    const run = async () => {
  1012→  1012→      setDialogEventsLoading(true);
  1013→  1013→      setDialogEvents([]);
  1014→  1014→      try {
  1015→  1015→        const res = await api.getEvents(sessionId, 500);
  1016→  1016→        if (cancelled) return;
  1017→  1017→        setDialogEvents(res.events ?? []);
  1018→  1018→      } catch (e: unknown) {
  1019→  1019→        if (cancelled) return;
  1020→  1020→        console.warn("[OffenseView] failed to load events for dialog:", e);
  1021→  1021→      } finally {
  1022→  1022→        if (!cancelled) setDialogEventsLoading(false);
  1023→  1023→      }
  1024→  1024→    };
  1025→  1025→    void run();
  1026→  1026→    return () => {
  1027→  1027→      cancelled = true;
  1028→  1028→    };
  1029→  1029→  }, [dialogScenarioId, mode, sessionId]);
  1030→  1030→
  1031→  1031→  // ---------------- Active scenarios (sorted) ----------------
  1032→  1032→  const scenarios = useMemo<OffenseScenario[]>(() => {
  1033→  1033→    const raw = mode === "historical" ? histScenarios : liveScenarios;
