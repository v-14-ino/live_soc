    65→function formatTime(ts: string | Date): string {
    66→  try {
    67→    const d = new Date(ts);
    68→    if (isNaN(d.getTime())) return "--:--:--";
    69→    return d.toLocaleTimeString("en-GB", { hour12: false });
    70→  } catch {
    71→    return "--:--:--";
    72→  }
    73→}
    74→
    75→function formatDateTime(ts: string | Date | null | undefined): string {
    76→  if (ts == null) return "—";
    77→  try {
    78→    const d = new Date(ts);
    79→    if (isNaN(d.getTime())) return "----/--/-- --:--:--";
    80→    const pad = (n: number) => String(n).padStart(2, "0");
    81→    return `${d.getFullYear()}/${pad(d.getMonth() + 1)}/${pad(d.getDate())} ${pad(d.getHours())}:${pad(d.getMinutes())}:${pad(d.getSeconds())}`;
    82→  } catch {
    83→    return "----/--/-- --:--:--";
    84→  }
    85→}

    87→function durationLabel(sec: number): string {
    88→  if (!Number.isFinite(sec) || sec < 0) return "—";
    89→  if (sec < 60) return `${sec}s`;
    90→  const m = Math.floor(sec / 60);
    91→  const s = sec % 60;
    92→  if (m < 60) return `${m}m ${s}s`;
    93→  const h = Math.floor(m / 60);
    94→  return `${h}h ${m % 60}m`;
    95→}

    97→function durationBetween(first: string | Date, last: string | Date | null | undefined): string {
    98→  if (!last) return "—";
    99→  try {
   100→    const a = new Date(first).getTime();
   101→    const b = new Date(last).getTime();
   102→    if (!isFinite(a) || !isFinite(b)) return "—";
   103→    const sec = Math.max(0, Math.round((b - a) / 1000));
   104→    return durationLabel(sec);
   105→  } catch {
   106→    return "—";
   107→  }
   108→}

   181→function normalizeAssessment(raw: unknown): AssessmentResult | null {
   182→  if (!raw) return null;
   183→  const r = raw as Record<string, unknown>;
   184→  const ports = Array.isArray(r.ports) ? r.ports : [];
   185→  const services = Array.isArray(r.services) ? r.services : [];
   186→  // Build a serviceId → name lookup so historical port rows (which store a
   187→  // foreign key in `serviceId`) can be resolved to a human-readable name.
   188→  const serviceNameById = new Map<string, string>();
   189→  for (const s of services) {
   190→    const sr = s as Record<string, unknown>;
   191→    if (sr.id && sr.name) serviceNameById.set(String(sr.id), String(sr.name));
   192→  }
   193→  return {
   194→    id: String(r.id),
   195→    targetId: String(r.targetId),
   196→    status: String(r.status ?? "pending"),
   197→    reachability: String(r.reachability ?? "unknown"),
   198→    latencyMs: r.latencyMs != null ? Number(r.latencyMs) : null,
   199→    hostname: (r.hostname as string | null) ?? null,
   200→    osGuess: (r.osGuess as string | null) ?? null,
   201→    startedAt: typeof r.startedAt === "string" ? r.startedAt : new Date(r.startedAt as Date).toISOString(),
   202→    completedAt: r.completedAt
   203→      ? typeof r.completedAt === "string"
   204→        ? r.completedAt
   205→        : new Date(r.completedAt as Date).toISOString()
   206→      : null,
   207→    ports: ports.map((p) => {
   208→      const pr = p as Record<string, unknown>;
   209→      const sid = pr.serviceId ? String(pr.serviceId) : null;
   210→      return {
   211→        id: String(pr.id),
   212→        number: Number(pr.number),
   213→        protocol: String(pr.protocol ?? "tcp"),
   214→        state: String(pr.state ?? "open"),
   215→        serviceName:
   216→          (pr.serviceName as string | undefined) ??
   217→          (sid ? serviceNameById.get(sid) : undefined),
   218→      };
   219→    }),
   220→    services: services.map((s) => ({
   221→      id: String((s as Record<string, unknown>).id),
   222→      name: String((s as Record<string, unknown>).name),
   223→      port: Number((s as Record<string, unknown>).port),
   224→      protocol: String((s as Record<string, unknown>).protocol ?? "tcp"),
   225→      product: ((s as Record<string, unknown>).product as string | null) ?? null,
   226→      version: ((s as Record<string, unknown>).version as string | null) ?? null,
   227→      extrainfo: ((s as Record<string, unknown>).extrainfo as string | null) ?? null,
   228→      method: String((s as Record<string, unknown>).method ?? "table"),
   229→      confidence: Number((s as Record<string, unknown>).confidence ?? 50),
   230→    })),
   231→  };
   232→}

   234→function parseRelatedEventIds(ids: string | string[] | null | undefined): string[] {
   235→  if (ids == null) return [];
   236→  if (Array.isArray(ids)) return ids.map(String).filter(Boolean);
   237→  const s = String(ids).trim();
   238→  if (!s) return [];
   239→  return s.split(",").map((x) => x.trim()).filter(Boolean);
   240→}
   241→
   242→function parseAttackPath(ap: string | string[] | null | undefined): string[] {
   243→  if (ap == null) return [];
   244→  if (Array.isArray(ap)) return ap.map(String).filter(Boolean);
   245→  const s = String(ap).trim();
   246→  if (!s) return [];
   247→  if (s.includes("->")) return s.split("->").map((x) => x.trim()).filter(Boolean);
   248→  if (s.includes("\n")) return s.split("\n").map((x) => x.trim()).filter(Boolean);
   249→  if (s.includes(",")) return s.split(",").map((x) => x.trim()).filter(Boolean);
   250→  return [s];
   251→}

   318→function portAccentColor(port: number): string {
   319→  if (port === 22 || port === 3389) return "var(--soc-medium)";
   320→  if ([80, 443, 8080, 8443].includes(port)) return "var(--soc-low)";
   321→  if ([3306, 5432, 6379, 1433, 27017, 1521].includes(port)) return "var(--soc-critical)";
   322→  if ([21, 23, 445, 139, 137].includes(port)) return "var(--soc-critical)";
   323→  if (port === 53) return "var(--soc-info)";
   324→  return "var(--foreground)";
   325→}

   327→function severityRank(s: Severity): number {
   328→  return SEVERITY_ORDER[s] ?? 0;
   329→}
   330→
   331→function sortOffense(list: OffenseScenario[]): OffenseScenario[] {
   332→  return list.slice().sort((a, b) => {
   333→    const sev = severityRank(b.severity) - severityRank(a.severity);
   334→    if (sev !== 0) return sev;
   335→    const ta = new Date(a.lastObserved).getTime() || 0;
   336→    const tb = new Date(b.lastObserved).getTime() || 0;
   337→    return tb - ta;
   338→  });
   339→}
   340→
   341→function sortDefense(list: DefenseScenario[]): DefenseScenario[] {
   342→  return list.slice().sort((a, b) => {
   343→    const sev = severityRank(b.priority) - severityRank(a.priority);
   344→    if (sev !== 0) return sev;
   345→    const ta = new Date(a.lastObserved).getTime() || 0;
   346→    const tb = new Date(b.lastObserved).getTime() || 0;
   347→    return tb - ta;
   348→  });
   349→}

   655→function Pagination({
   656→  page,
   657→  pageSize,
   658→  total,
   659→  onPage,
   660→}: {
   661→  page: number;
   662→  pageSize: number;
   663→  total: number;
   664→  onPage: (p: number) => void;
   665→}) {
   666→  const totalPages = Math.max(1, Math.ceil(total / pageSize));
   667→  const from = total === 0 ? 0 : (page - 1) * pageSize + 1;
   668→  const to = Math.min(total, page * pageSize);
   669→  return (
   670→    <div className="flex items-center justify-between gap-2">
   671→      <span className="font-mono-data text-[10px] text-muted-foreground">
   672→        Showing <span className="text-foreground/90">{from}–{to}</span> of{" "}
   673→        <span className="text-foreground/90">{total}</span> · Page{" "}
   674→        <span className="text-foreground/90">{page}</span> /{" "}
   675→        <span className="text-foreground/90">{totalPages}</span>
   676→      </span>
   677→      <div className="flex items-center gap-1">
   678→        <Button
   679→          size="sm"
   680→          variant="outline"
   681→          disabled={page <= 1}
   682→          onClick={() => onPage(page - 1)}
   683→          className="h-7 gap-1 px-2 text-[11px]"
   684→        >
   685→          <ChevronLeft className="h-3 w-3" />
   686→          Prev
   687→        </Button>
   688→        <Button
   689→          size="sm"
   690→          variant="outline"
   691→          disabled={page >= totalPages}
   692→          onClick={() => onPage(page + 1)}
   693→          className="h-7 gap-1 px-2 text-[11px]"
   694→        >
   695→          Next
   696→          <ChevronRight className="h-3 w-3" />
   697→        </Button>
   698→      </div>
   699→    </div>
   700→  );
   701→}
