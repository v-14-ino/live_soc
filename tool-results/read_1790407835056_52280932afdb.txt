    69→function formatTime(ts: string | Date): string {
    70→  try {
    71→    const d = new Date(ts);
    72→    if (isNaN(d.getTime())) return "--:--:--";
    73→    return d.toLocaleTimeString("en-GB", { hour12: false });
    74→  } catch {
    75→    return "--:--:--";
    76→  }
    77→}
    78→
    79→function formatDateTime(ts: string | Date | null | undefined): string {
    80→  if (ts == null) return "—";
    81→  try {
    82→    const d = new Date(ts);
    83→    if (isNaN(d.getTime())) return "----/--/-- --:--:--";
    84→    const pad = (n: number) => String(n).padStart(2, "0");
    85→    return `${d.getFullYear()}/${pad(d.getMonth() + 1)}/${pad(d.getDate())} ${pad(d.getHours())}:${pad(d.getMinutes())}:${pad(d.getSeconds())}`;
    86→  } catch {
    87→    return "----/--/-- --:--:--";
    88→  }
    89→}

    91→function durationLabel(sec: number): string {
    92→  if (!Number.isFinite(sec) || sec < 0) return "—";
    93→  if (sec < 60) return `${sec}s`;
    94→  const m = Math.floor(sec / 60);
    95→  const s = sec % 60;
    96→  if (m < 60) return `${m}m ${s}s`;
    97→  const h = Math.floor(m / 60);
    98→  return `${h}h ${m % 60}m`;
    99→}

   101→function durationBetween(first: string | Date, last: string | Date | null | undefined): string {
   102→  if (!last) return "—";
   103→  try {
   104→    const a = new Date(first).getTime();
   105→    const b = new Date(last).getTime();
   106→    if (!isFinite(a) || !isFinite(b)) return "—";
   107→    const sec = Math.max(0, Math.round((b - a) / 1000));
   108→    return durationLabel(sec);
   109→  } catch {
   110→    return "—";
   111→  }
   112→}

   185→function normalizeAssessment(raw: unknown): AssessmentResult | null {
   186→  if (!raw) return null;
   187→  const r = raw as Record<string, unknown>;
   188→  const ports = Array.isArray(r.ports) ? r.ports : [];
   189→  const services = Array.isArray(r.services) ? r.services : [];
   190→  // Build a serviceId → name lookup so historical port rows (which store a
   191→  // foreign key in `serviceId`) can be resolved to a human-readable name.
   192→  const serviceNameById = new Map<string, string>();
   193→  for (const s of services) {
   194→    const sr = s as Record<string, unknown>;
   195→    if (sr.id && sr.name) serviceNameById.set(String(sr.id), String(sr.name));
   196→  }
   197→  return {
   198→    id: String(r.id),
   199→    targetId: String(r.targetId),
   200→    status: String(r.status ?? "pending"),
   201→    reachability: String(r.reachability ?? "unknown"),
   202→    latencyMs: r.latencyMs != null ? Number(r.latencyMs) : null,
   203→    hostname: (r.hostname as string | null) ?? null,
   204→    osGuess: (r.osGuess as string | null) ?? null,
   205→    startedAt: typeof r.startedAt === "string" ? r.startedAt : new Date(r.startedAt as Date).toISOString(),
   206→    completedAt: r.completedAt
   207→      ? typeof r.completedAt === "string"
   208→        ? r.completedAt
   209→        : new Date(r.completedAt as Date).toISOString()
   210→      : null,
   211→    ports: ports.map((p) => {
   212→      const pr = p as Record<string, unknown>;
   213→      const sid = pr.serviceId ? String(pr.serviceId) : null;
   214→      return {
   215→        id: String(pr.id),
   216→        number: Number(pr.number),
   217→        protocol: String(pr.protocol ?? "tcp"),
   218→        state: String(pr.state ?? "open"),
   219→        serviceName:
   220→          (pr.serviceName as string | undefined) ??
   221→          (sid ? serviceNameById.get(sid) : undefined),
   222→      };
   223→    }),
   224→    services: services.map((s) => ({
   225→      id: String((s as Record<string, unknown>).id),
   226→      name: String((s as Record<string, unknown>).name),
   227→      port: Number((s as Record<string, unknown>).port),
   228→      protocol: String((s as Record<string, unknown>).protocol ?? "tcp"),
   229→      product: ((s as Record<string, unknown>).product as string | null) ?? null,
   230→      version: ((s as Record<string, unknown>).version as string | null) ?? null,
   231→      extrainfo: ((s as Record<string, unknown>).extrainfo as string | null) ?? null,
   232→      method: String((s as Record<string, unknown>).method ?? "table"),
   233→      confidence: Number((s as Record<string, unknown>).confidence ?? 50),
   234→    })),
   235→  };
   236→}

   238→function parseRelatedEventIds(ids: string | string[] | null | undefined): string[] {
   239→  if (ids == null) return [];
   240→  if (Array.isArray(ids)) return ids.map(String).filter(Boolean);
   241→  const s = String(ids).trim();
   242→  if (!s) return [];
   243→  return s.split(",").map((x) => x.trim()).filter(Boolean);
   244→}
   245→
   246→function parseAttackPath(ap: string | string[] | null | undefined): string[] {
   247→  if (ap == null) return [];
   248→  if (Array.isArray(ap)) return ap.map(String).filter(Boolean);
   249→  const s = String(ap).trim();
   250→  if (!s) return [];
   251→  if (s.includes("->")) return s.split("->").map((x) => x.trim()).filter(Boolean);
   252→  if (s.includes("\n")) return s.split("\n").map((x) => x.trim()).filter(Boolean);
   253→  if (s.includes(",")) return s.split(",").map((x) => x.trim()).filter(Boolean);
   254→  return [s];
   255→}

   322→function portAccentColor(port: number): string {
   323→  if (port === 22 || port === 3389) return "var(--soc-medium)";
   324→  if ([80, 443, 8080, 8443].includes(port)) return "var(--soc-low)";
   325→  if ([3306, 5432, 6379, 1433, 27017, 1521].includes(port)) return "var(--soc-critical)";
   326→  if ([21, 23, 445, 139, 137].includes(port)) return "var(--soc-critical)";
   327→  if (port === 53) return "var(--soc-info)";
   328→  return "var(--foreground)";
   329→}

   331→function severityRank(s: Severity): number {
   332→  return SEVERITY_ORDER[s] ?? 0;
   333→}
   334→
   335→function sortOffense(list: OffenseScenario[]): OffenseScenario[] {
   336→  return list.slice().sort((a, b) => {
   337→    const sev = severityRank(b.severity) - severityRank(a.severity);
   338→    if (sev !== 0) return sev;
   339→    const ta = new Date(a.lastObserved).getTime() || 0;
   340→    const tb = new Date(b.lastObserved).getTime() || 0;
   341→    return tb - ta;
   342→  });
   343→}
   344→
   345→function sortDefense(list: DefenseScenario[]): DefenseScenario[] {
   346→  return list.slice().sort((a, b) => {
   347→    const sev = severityRank(b.priority) - severityRank(a.priority);
   348→    if (sev !== 0) return sev;
   349→    const ta = new Date(a.lastObserved).getTime() || 0;
   350→    const tb = new Date(b.lastObserved).getTime() || 0;
   351→    return tb - ta;
   352→  });
   353→}

   659→function Pagination({
   660→  page,
   661→  pageSize,
   662→  total,
   663→  onPage,
   664→}: {
   665→  page: number;
   666→  pageSize: number;
   667→  total: number;
   668→  onPage: (p: number) => void;
   669→}) {
   670→  const totalPages = Math.max(1, Math.ceil(total / pageSize));
   671→  const from = total === 0 ? 0 : (page - 1) * pageSize + 1;
   672→  const to = Math.min(total, page * pageSize);
   673→  return (
   674→    <div className="flex items-center justify-between gap-2">
   675→      <span className="font-mono-data text-[10px] text-muted-foreground">
   676→        Showing <span className="text-foreground/90">{from}–{to}</span> of{" "}
   677→        <span className="text-foreground/90">{total}</span> · Page{" "}
   678→        <span className="text-foreground/90">{page}</span> /{" "}
   679→        <span className="text-foreground/90">{totalPages}</span>
   680→      </span>
   681→      <div className="flex items-center gap-1">
   682→        <Button
   683→          size="sm"
   684→          variant="outline"
   685→          disabled={page <= 1}
   686→          onClick={() => onPage(page - 1)}
   687→          className="h-7 gap-1 px-2 text-[11px]"
   688→        >
   689→          <ChevronLeft className="h-3 w-3" />
   690→          Prev
   691→        </Button>
   692→        <Button
   693→          size="sm"
   694→          variant="outline"
   695→          disabled={page >= totalPages}
   696→          onClick={() => onPage(page + 1)}
   697→          className="h-7 gap-1 px-2 text-[11px]"
   698→        >
   699→          Next
   700→          <ChevronRight className="h-3 w-3" />
   701→        </Button>
   702→      </div>
   703→    </div>

  1733→export function HistoryView() {
  1734→  const [sessions, setSessions] = useState<MonitoringSessionInfo[]>([]);
  1735→  const [total, setTotal] = useState(0);
  1736→  const [page, setPage] = useState(1);
  1737→  const [pageSize, setPageSize] = useState(20);
  1738→  const [statusFilter, setStatusFilter] = useState("all");
  1739→  const [targetSearch, setTargetSearch] = useState("");
  1740→  const [loading, setLoading] = useState(false);
  1741→  const [error, setError] = useState<string | null>(null);
  1742→  const [selected, setSelected] = useState<MonitoringSessionInfo | null>(null);
  1743→  const [dialogOpen, setDialogOpen] = useState(false);
  1744→
  1745→  // Debounce target search
  1746→  const [searchInput, setSearchInput] = useState("");
  1747→  useEffect(() => {
  1748→    const id = setTimeout(() => setTargetSearch(searchInput.trim()), 350);
  1749→    return () => clearTimeout(id);
  1750→  }, [searchInput]);
  1751→
  1752→  // Reset to page 1 when filters change
  1753→  useEffect(() => {
  1754→    setPage(1);
  1755→  }, [statusFilter, targetSearch, pageSize]);
  1756→
  1757→  const fetchHistory = async () => {
  1758→    setLoading(true);
  1759→    setError(null);
  1760→    try {
  1761→      const res: HistoryListResponse = await api.getHistory({
  1762→        page,
  1763→        pageSize,
  1764→        status: statusFilter === "all" ? undefined : statusFilter,
  1765→        target: targetSearch || undefined,
  1766→      });
  1767→      const normalized = (res.sessions ?? []).map(normalizeSession);
  1768→      setSessions(normalized);
  1769→      setTotal(res.total ?? 0);
  1770→    } catch (e) {
  1771→      const msg = e instanceof Error ? e.message : "Failed to load history.";
  1772→      setError(msg);
  1773→      toast.error(msg);
  1774→    } finally {
  1775→      setLoading(false);
  1776→    }
  1777→  };
  1778→
  1779→  useEffect(() => {
  1780→    void fetchHistory();
  1781→  }, [page, pageSize, statusFilter, targetSearch]);
  1782→
  1783→  const handleOpen = (s: MonitoringSessionInfo) => {
  1784→    setSelected(s);
  1785→    setDialogOpen(true);
  1786→  };
  1787→
  1788→  const totals = useMemo(() => {
  1789→    let totalEvents = 0;
  1790→    let totalAlerts = 0;
  1791→    for (const s of sessions) {
  1792→      totalEvents += s.eventCount;
  1793→      totalAlerts += s.alertCount;
  1794→    }
  1795→    return { totalEvents, totalAlerts };
  1796→  }, [sessions]);
  1797→
  1798→  const showEmpty = !loading && !error && sessions.length === 0;
  1799→
  1800→  return (
  1801→    <div className="flex h-full flex-col overflow-hidden">
  1802→      <HistoryHeader
  1803→        onRefresh={() => void fetchHistory()}
  1804→        refreshing={loading}
  1805→        statusFilter={statusFilter}
  1806→        setStatusFilter={setStatusFilter}
  1807→        targetSearch={searchInput}
  1808→        setTargetSearch={setSearchInput}
  1809→        pageSize={pageSize}
  1810→        setPageSize={setPageSize}
  1811→      />
  1812→
  1813→      <div className="shrink-0 border-b border-border/40 px-3 py-3">
  1814→        <SummaryRow
  1815→          total={total}
  1816→          totalEvents={totals.totalEvents}
  1817→          totalAlerts={totals.totalAlerts}
  1818→        />
  1819→      </div>
  1820→
  1821→      <div className="soc-scrollbar min-h-0 flex-1 overflow-y-auto overflow-x-hidden">
  1822→        {showEmpty ? (
  1823→          <NoHistoryEmptyState />
  1824→        ) : (
  1825→          <div className="flex flex-col gap-3 p-3">
  1826→            {error && (
  1827→              <div className="flex items-center gap-2 rounded-md border border-[color:var(--soc-critical)]/40 bg-[color:var(--soc-critical)]/10 px-3 py-2 text-[11px] text-[color:var(--soc-critical)]">
  1828→                <XCircle className="h-3.5 w-3.5" />
  1829→                {error}
  1830→              </div>
  1831→            )}
  1832→            <SessionsTable
  1833→              sessions={sessions}
  1834→              loading={loading}
  1835→              onOpen={handleOpen}
  1836→            />
  1837→            {sessions.length > 0 && (
  1838→              <Pagination
  1839→                page={page}
  1840→                pageSize={pageSize}
  1841→                total={total}
  1842→                onPage={setPage}

  1844→            )}
  1845→          </div>
  1846→        )}
  1847→      </div>
  1848→
  1849→      <DetailDialog
  1850→        session={selected}
  1851→        open={dialogOpen}
  1852→        onOpenChange={setDialogOpen}
  1853→      />
