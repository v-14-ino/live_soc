    69→function formatTime(ts: string): string {
    70→  try {
    71→    const d = new Date(ts);
    72→    if (isNaN(d.getTime())) return "--:--:--";
    73→    return d.toLocaleTimeString("en-GB", { hour12: false });
    74→  } catch {
    75→    return "--:--:--";
    76→  }
    77→}

    79→function portAccentColor(port: number): string {
    80→  if (port === 22 || port === 3389) return "var(--soc-medium)";
    81→  if ([80, 443, 8080, 8443].includes(port)) return "var(--soc-low)";
    82→  if ([3306, 5432, 6379, 1433, 27017, 1521].includes(port)) return "var(--soc-critical)";
    83→  if ([21, 23, 445, 139, 137].includes(port)) return "var(--soc-critical)";
    84→  if (port === 53) return "var(--soc-info)";
    85→  return "var(--foreground)";
    86→}

   365→function TargetControlBar() {
   366→  const sessionId = useAppStore((s) => s.sessionId);
   367→  const status = useAppStore((s) => s.status);
   368→  const targetAddress = useAppStore((s) => s.targetAddress);
   369→  const connected = useAppStore((s) => s.connected);
   370→  const mode = useAppStore((s) => s.mode);
   371→  const startSession = useAppStore((s) => s.startSession);
   372→  const stopSession = useAppStore((s) => s.stopSession);
   373→  const resetSession = useAppStore((s) => s.resetSession);
   374→  const setStatus = useAppStore((s) => s.setStatus);
   375→  const setKpi = useAppStore((s) => s.setKpi);
   376→  const setNetworkActivity = useAppStore((s) => s.setNetworkActivity);
   377→  const setOffenseScenarios = useAppStore((s) => s.setOffenseScenarios);
   378→  const setDefenseScenarios = useAppStore((s) => s.setDefenseScenarios);
   379→
   380→  const [targetInput, setTargetInput] = useState("192.168.1.100");
   381→  const [modeInput, setModeInput] = useState<"demo" | "live">("demo");
   382→  const [busy, setBusy] = useState(false);
   383→
   384→  const isActive = statusIsActive(status);
   385→
   386→  const handleStart = async () => {
   387→    const t = targetInput.trim();
   388→    if (!t) {
   389→      toast.error("Enter a target IP or domain.");
   390→      return;
   391→    }
   392→    setBusy(true);
   393→    setStatus("initializing");
   394→    try {
   395→      const res = await api.startMonitoring(t, modeInput);
   396→      const resolvedTarget = res.session.targetAddress || t;
   397→      startSession({
   398→        sessionId: res.sessionId,
   399→        targetAddress: resolvedTarget,
   400→        mode: res.session.mode,
   401→        assessment: res.assessment,
   402→      });
   403→      toast.success(`Monitoring started on ${resolvedTarget}`);
   404→      // Seed initial status snapshot (WS will stream updates afterwards)
   405→      try {
   406→        const st = await api.getStatus(res.sessionId);
   407→        if (st.kpi) setKpi(st.kpi);
   408→        if (st.networkActivity) setNetworkActivity(st.networkActivity);
   409→        if (st.offenseScenarios?.length) setOffenseScenarios(st.offenseScenarios);
   410→        if (st.defenseScenarios?.length) setDefenseScenarios(st.defenseScenarios);
   411→      } catch {
   412→        /* best-effort seeding; WS will provide live updates */
   413→      }
   414→    } catch (err) {
   415→      const e = err as Error & { status?: number; code?: string };
   416→      if (e.code === "INVALID_TARGET") {
   417→        toast.error("Target is unreachable.");
   418→      } else if (e.code === "START_FAILED") {
   419→        toast.error("Initial assessment failed.");
   420→      } else if (e.code === "UNREACHABLE" || !e.status) {
   421→        toast.error("Unable to connect to monitoring service.");
   422→      } else {
   423→        toast.error(e.message || "Unable to start monitoring.");
   424→      }
   425→      setStatus("ready");
   426→    } finally {
   427→      setBusy(false);
   428→    }
   429→  };
   430→
   431→  const handleStop = async () => {
   432→    if (!sessionId) return;
   433→    setBusy(true);
   434→    try {
   435→      await api.stopMonitoring(sessionId);
   436→      stopSession();
   437→      toast.success("Monitoring session saved to history.");
   438→      window.setTimeout(() => resetSession(), 1000);
   439→    } catch (err) {
   440→      const e = err as Error & { status?: number; code?: string };
   441→      if (e.code === "UNREACHABLE" || !e.status) {
   442→        toast.error("Unable to connect to monitoring service.");
   443→      } else {
   444→        toast.error(e.message || "Failed to stop monitoring.");
   445→      }
   446→      stopSession();
   447→      window.setTimeout(() => resetSession(), 1000);
   448→    } finally {
   449→      setBusy(false);
   450→    }
   451→  };
   452→
   453→  return (
   454→    <div className="flex shrink-0 flex-wrap items-center gap-2 border-b border-border/60 bg-card/40 px-3 py-2 backdrop-blur-md">
   455→      <div className="flex items-center gap-2">
   456→        <span className="font-mono-data text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   457→          Target
   458→        </span>
   459→        <Input
   460→          value={targetInput}
   461→          onChange={(e) => setTargetInput(e.target.value)}
   462→          placeholder="IP or domain"
   463→          disabled={isActive || busy}
   464→          className="h-8 w-[200px] font-mono-data text-sm sm:w-[280px]"
   465→          onKeyDown={(e) => {
   466→            if (e.key === "Enter" && !isActive && !busy) handleStart();
   467→          }}
   468→          aria-label="Monitoring target IP or domain"
   469→        />
   470→      </div>
   471→
   472→      <Select
   473→        value={modeInput}
   474→        onValueChange={(v) => setModeInput(v as "demo" | "live")}
   475→        disabled={isActive || busy}
   476→      >
   477→        <SelectTrigger size="sm" className="h-8 w-full font-mono-data text-xs sm:w-[260px]">
   478→          <SelectValue />
   479→        </SelectTrigger>
   480→        <SelectContent>
   481→          <SelectItem value="demo">Demo Mode (Simulated Telemetry)</SelectItem>
   482→          <SelectItem value="live">Live Mode (Real Telemetry)</SelectItem>
   483→        </SelectContent>
   484→      </Select>
   485→
   486→      {!isActive ? (
   487→        <Button
   488→          onClick={handleStart}
   489→          disabled={busy}
   490→          size="sm"
   491→          className="h-8 gap-1.5"
   492→        >
   493→          <Activity className="h-3.5 w-3.5" />
   494→          {busy ? "STARTING…" : "START MONITORING"}
   495→        </Button>
   496→      ) : (
   497→        <Button
   498→          onClick={handleStop}
   499→          disabled={busy}
   500→          size="sm"
   501→          variant="destructive"
   502→          className="h-8 gap-1.5"
   503→        >
   504→          <Square className="h-3.5 w-3.5 fill-current" />
   505→          {busy ? "STOPPING…" : "STOP MONITORING"}
   506→        </Button>
   507→      )}
   508→
   509→      {isActive && (
   510→        <div className="flex flex-wrap items-center gap-3 pl-1">
   511→          <div className="flex items-center gap-1.5">
   512→            <span className="font-mono-data text-[10px] uppercase tracking-wider text-muted-foreground">
   513→              Active
   514→            </span>
   515→            <span className="font-mono-data text-xs font-semibold">
   516→              {targetAddress}
   517→            </span>
   518→          </div>
   519→          <StatusDot
   520→            status={status}
   521→            pulse={isActive}
   522→            label={status.toUpperCase()}
   523→            className="text-[10px]"
   524→          />
   525→          <StatusDot
   526→            status={connected ? "connected" : "disconnected"}
   527→            pulse={connected}
   528→            label={connected ? "LIVE" : "RECONNECTING"}
   529→            className="text-[10px]"
   530→          />
   531→          {mode === "demo" && (
   532→            <span className="rounded-sm border border-[color:var(--soc-medium)]/40 bg-[color:var(--soc-medium)]/10 px-1.5 py-0.5 font-mono-data text-[9px] font-bold uppercase tracking-wider text-[color:var(--soc-medium)]">
   533→              Demo
   534→            </span>
   535→          )}
   536→        </div>
   537→      )}
   538→
   539→      <div className="ml-auto hidden items-center md:flex">
   540→        <AuthWarning />
   541→      </div>
   542→    </div>
   543→  );
   544→}
