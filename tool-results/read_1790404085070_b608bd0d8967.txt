    71→function formatTime(ts: string): string {
    72→  try {
    73→    const d = new Date(ts);
    74→    if (isNaN(d.getTime())) return "--:--:--";
    75→    return d.toLocaleTimeString("en-GB", { hour12: false });
    76→  } catch {
    77→    return "--:--:--";
    78→  }
    79→}

    81→function portAccentColor(port: number): string {
    82→  if (port === 22 || port === 3389) return "var(--soc-medium)";
    83→  if ([80, 443, 8080, 8443].includes(port)) return "var(--soc-low)";
    84→  if ([3306, 5432, 6379, 1433, 27017, 1521].includes(port)) return "var(--soc-critical)";
    85→  if ([21, 23, 445, 139, 137].includes(port)) return "var(--soc-critical)";
    86→  if (port === 53) return "var(--soc-info)";
    87→  return "var(--foreground)";
    88→}

   367→function TargetControlBar() {
   368→  const sessionId = useAppStore((s) => s.sessionId);
   369→  const status = useAppStore((s) => s.status);
   370→  const targetAddress = useAppStore((s) => s.targetAddress);
   371→  const connected = useAppStore((s) => s.connected);
   372→  const mode = useAppStore((s) => s.mode);
   373→  const startSession = useAppStore((s) => s.startSession);
   374→  const stopSession = useAppStore((s) => s.stopSession);
   375→  const resetSession = useAppStore((s) => s.resetSession);
   376→  const setStatus = useAppStore((s) => s.setStatus);
   377→  const setKpi = useAppStore((s) => s.setKpi);
   378→  const setNetworkActivity = useAppStore((s) => s.setNetworkActivity);
   379→  const setOffenseScenarios = useAppStore((s) => s.setOffenseScenarios);
   380→  const setDefenseScenarios = useAppStore((s) => s.setDefenseScenarios);
   381→
   382→  const [targetInput, setTargetInput] = useState("192.168.1.100");
   383→  const [modeInput, setModeInput] = useState<"demo" | "live">("demo");
   384→  const [busy, setBusy] = useState(false);
   385→
   386→  const isActive = statusIsActive(status);
   387→
   388→  const handleStart = async () => {
   389→    const t = targetInput.trim();
   390→    if (!t) {
   391→      toast.error("Enter a target IP or domain.");
   392→      return;
   393→    }
   394→    setBusy(true);
   395→    setStatus("initializing");
   396→    try {
   397→      const res = await api.startMonitoring(t, modeInput);
   398→      const resolvedTarget = res.session.targetAddress || t;
   399→      startSession({
   400→        sessionId: res.sessionId,
   401→        targetAddress: resolvedTarget,
   402→        mode: res.session.mode,
   403→        assessment: res.assessment,
   404→      });
   405→      toast.success(`Monitoring started on ${resolvedTarget}`);
   406→      // Seed initial status snapshot (WS will stream updates afterwards)
   407→      try {
   408→        const st = await api.getStatus(res.sessionId);
   409→        if (st.kpi) setKpi(st.kpi);
   410→        if (st.networkActivity) setNetworkActivity(st.networkActivity);
   411→        if (st.offenseScenarios?.length) setOffenseScenarios(st.offenseScenarios);
   412→        if (st.defenseScenarios?.length) setDefenseScenarios(st.defenseScenarios);
   413→      } catch {
   414→        /* best-effort seeding; WS will provide live updates */
   415→      }
   416→    } catch (err) {
   417→      const e = err as Error & { status?: number; code?: string };
   418→      if (e.code === "INVALID_TARGET") {
   419→        toast.error("Target is unreachable.");
   420→      } else if (e.code === "START_FAILED") {
   421→        toast.error("Initial assessment failed.");
   422→      } else if (e.code === "UNREACHABLE" || !e.status) {
   423→        toast.error("Unable to connect to monitoring service.");
   424→      } else {
   425→        toast.error(e.message || "Unable to start monitoring.");
   426→      }
   427→      setStatus("ready");
   428→    } finally {
   429→      setBusy(false);
   430→    }
   431→  };
   432→
   433→  const handleStop = async () => {
   434→    if (!sessionId) return;
   435→    setBusy(true);
   436→    try {
   437→      await api.stopMonitoring(sessionId);
   438→      stopSession();
   439→      toast.success("Monitoring session saved to history.");
   440→      window.setTimeout(() => resetSession(), 1000);
   441→    } catch (err) {
   442→      const e = err as Error & { status?: number; code?: string };
   443→      if (e.code === "UNREACHABLE" || !e.status) {
   444→        toast.error("Unable to connect to monitoring service.");
   445→      } else {
   446→        toast.error(e.message || "Failed to stop monitoring.");
   447→      }
   448→      stopSession();
   449→      window.setTimeout(() => resetSession(), 1000);
   450→    } finally {
   451→      setBusy(false);
   452→    }
   453→  };
   454→
   455→  return (
   456→    <div className="flex shrink-0 flex-wrap items-center gap-2 border-b border-border/60 bg-card/40 px-3 py-2 backdrop-blur-md">
   457→      <div className="flex items-center gap-2">
   458→        <span className="font-mono-data text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   459→          Target
   460→        </span>
   461→        <Input
   462→          value={targetInput}
   463→          onChange={(e) => setTargetInput(e.target.value)}
   464→          placeholder="IP or domain"
   465→          disabled={isActive || busy}
   466→          className="h-8 w-[200px] font-mono-data text-sm sm:w-[280px]"
   467→          onKeyDown={(e) => {
   468→            if (e.key === "Enter" && !isActive && !busy) handleStart();
   469→          }}
   470→          aria-label="Monitoring target IP or domain"
   471→        />
   472→      </div>
   473→
   474→      <Select
   475→        value={modeInput}
   476→        onValueChange={(v) => setModeInput(v as "demo" | "live")}
   477→        disabled={isActive || busy}
   478→      >
   479→        <SelectTrigger size="sm" className="h-8 w-full font-mono-data text-xs sm:w-[260px]">
   480→          <SelectValue />
   481→        </SelectTrigger>
   482→        <SelectContent>
   483→          <SelectItem value="demo">Demo Mode (Simulated Telemetry)</SelectItem>
   484→          <SelectItem value="live">Live Mode (Real Telemetry)</SelectItem>
   485→        </SelectContent>
   486→      </Select>
   487→
   488→      {!isActive ? (
   489→        <Button
   490→          onClick={handleStart}
   491→          disabled={busy}
   492→          size="sm"
   493→          className="h-8 gap-1.5"
   494→        >
   495→          <Activity className="h-3.5 w-3.5" />
   496→          {busy ? "STARTING…" : "START MONITORING"}
   497→        </Button>
   498→      ) : (
   499→        <Button
   500→          onClick={handleStop}
   501→          disabled={busy}
   502→          size="sm"
   503→          variant="destructive"
   504→          className="h-8 gap-1.5"
   505→        >
   506→          <Square className="h-3.5 w-3.5 fill-current" />
   507→          {busy ? "STOPPING…" : "STOP MONITORING"}
   508→        </Button>
   509→      )}
   510→
   511→      {isActive && (
   512→        <div className="flex flex-wrap items-center gap-3 pl-1">
   513→          <div className="flex items-center gap-1.5">
   514→            <span className="font-mono-data text-[10px] uppercase tracking-wider text-muted-foreground">
   515→              Active
   516→            </span>
   517→            <span className="font-mono-data text-xs font-semibold">
   518→              {targetAddress}
   519→            </span>
   520→          </div>
   521→          <StatusDot
   522→            status={status}
   523→            pulse={isActive}
   524→            label={status.toUpperCase()}
   525→            className="text-[10px]"
   526→          />
   527→          <StatusDot
   528→            status={connected ? "connected" : "disconnected"}
   529→            pulse={connected}
   530→            label={connected ? "LIVE" : "RECONNECTING"}
   531→            className="text-[10px]"
   532→          />
   533→          {mode === "demo" && (
   534→            <span className="rounded-sm border border-[color:var(--soc-medium)]/40 bg-[color:var(--soc-medium)]/10 px-1.5 py-0.5 font-mono-data text-[9px] font-bold uppercase tracking-wider text-[color:var(--soc-medium)]">
   535→              Demo
   536→            </span>
   537→          )}
   538→        </div>
   539→      )}
   540→
   541→      <div className="ml-auto hidden items-center md:flex">
   542→        <AuthWarning />
   543→      </div>
   544→    </div>
   545→  );
   546→}

   552→function AssessmentPanel() {

   574→            {open ? (
   575→              <ChevronDown className="h-4 w-4 shrink-0 text-muted-foreground" />

   577→              <ChevronRight className="h-4 w-4 shrink-0 text-muted-foreground" />
   578→            )}
   579→            <span className="font-mono-data text-xs font-semibold uppercase tracking-wider">
   580→              Initial Assessment

   583→              {assessment.reachability} · {portCount} open port{portCount === 1 ? "" : "s"}
